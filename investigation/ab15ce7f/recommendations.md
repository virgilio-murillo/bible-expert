### Phase 4: Enrich OT Xref Data (Hebrew + English for OT→OT refs)

In `server.py` or wherever xrefs are fetched, add Hebrew (WLC) and English (KJV) text for each cross-reference:

```python
def enrich_xref(xref: dict, db) -> dict:
    """Add Hebrew and English text to cross-references."""
    ref = xref["ref"]  # e.g. "Proverbs 15:5"
    book, ch_v = parse_ref(ref)
    text = xref.get("text", {})
    if not isinstance(text, dict):
        text = {"es": str(text)}
    
    # Add Hebrew (WLC) for OT refs
    if is_ot_book(book) and not text.get("heb"):
        hebrew = db.execute("SELECT text FROM wlc WHERE book=? AND chapter=? AND verse=?", ...).fetchone()
        if hebrew:
            text["heb"] = hebrew[0]
    
    # Add LXX for OT refs  
    if is_ot_book(book) and not text.get("lxx"):
        lxx = db.execute("SELECT text FROM lxx WHERE book=? AND chapter=? AND verse=?", ...).fetchone()
        if lxx:
            text["lxx"] = lxx[0]
    
    # Add English (KJV) for all refs
    if not text.get("en"):
        kjv = db.execute("SELECT text FROM kjv WHERE book=? AND chapter=? AND verse=?", ...).fetchone()
        if kjv:
            text["en"] = kjv[0]
    
    xref["text"] = text
    return xref
```
